Practical 2: detect edge using sobel and canny compare result
import cv2
import matplotlib.pyplot as plt

img_rgb=cv2.cvtColor(img,cv2.COLOR_BGR2RGB)

plt.imshow(img_rgb)
plt.title("original image")

plt.imshow(sobel,cmap="gray")
plt.title("sobel edge detection")
plt.axis("off")
plt.subplot(1,3,3)
plt.imshow(canny,cmap="gray")
plt.title("canny edge detection")
plt.axis("off")
plt.show()

Practical 4: Match keypoints between two images using SIFT + Ransac
import cv2
import numpy as np
import matplotlib.pyplot as plt
img1=cv2.imread(r"C:\Users\LAB202\Pictures\images.jpg")
img2=cv2.imread(r"image1.png")
gray1=cv2.cvtColor(img1,cv2.COLOR_BGR2GRAY)
gray2=cv2.cvtColor(img2,cv2.COLOR_BGR2GRAY)
sift=cv2.SIFT_create()
kp1, des1=sift.detectAndCompute(gray1,None)
kp2, des2=sift.detectAndCompute(gray2,None)
bf=cv2.BFMatcher()
matches=bf.match(des1,des2)
matches=sorted(matches,key=lambda x: x.distance)
src_pts= np.float32([kp1[m.queryIdx].pt for m in matches]).reshape(-1,1,2)
dst_pts= np.float32([kp2[m.trainIdx].pt for m in matches]).reshape(-1,1,2)
H, mask=cv2.findHomography(src_pts,dst_pts,cv2.RANSAC,5.0)
matchesMask=mask.ravel().tolist()
result=cv2.drawMatches(
img1,
kp1,
img2,
kp2,
matches,
None,
matchesMask=matchesMask,
flags=cv2.DrawMatchesFlags_NOT_DRAW_SINGLE_POINTS
)
plt.figure(figsize=(15,8))
plt.imshow(cv2.cvtColor(result,cv2.COLOR_BGR2RGB))
plt.title("SIFT + RANSAC Matching")
plt.axis("off")
plt.show()

Practical 5: Segment an image using k-means color clustering
import cv2
import numpy as np
import matplotlib.pyplot as plt
image = cv2.imread("")
image_rgb = cv2.cvtColor(image, cv2.COLOR_BGR2RGB)
pixels = image_rgb.reshape((-1, 3))
pixels = np.float32(pixels)
K = 4
criteria = (
    cv2.TERM_CRITERIA_EPS + cv2.TERM_CRITERIA_MAX_ITER,
    100,
    0.2
)
compactness, labels, centers = cv2.kmeans(
    pixels,
    K,
    None,
    criteria,
    10,
    cv2.KMEANS_RANDOM_CENTERS
)
centers = np.uint8(centers)
segmented_data = centers[labels.flatten()]
segmented_image = segmented_data.reshape(image_rgb.shape)
plt.figure(figsize=(10, 5))
plt.subplot(1, 2, 1)
plt.imshow(image_rgb)
plt.title("Original Image")
plt.axis("off")
plt.subplot(1, 2, 2)
plt.imshow(segmented_image)
plt.title("K-Means Segmented Image")
plt.axis("off")
plt.show()

Practical 6: Implement optical flow on a video clip with Lucas Kanade
import cv2
import numpy as np
cap=cv2.VideoCapture(r"")
ret, old_frame=cap.read()
if not ret:
    print("cannot open video")
    exit()
old_gray=cv2.cvtColor(old_frame, cv2.COLOR_BGR2GRAY)
old_points=cv2.goodFeaturesToTrack(
    old_gray,
    maxCorners=100,
    qualityLevel=0.3,
    minDistance=7,
    blockSize=7
)
lk_params=dict(
    winSize=(15,15),
    maxLevel=2,
    criteria=(
        cv2.TERM_CRITERIA_EPS +
        cv2.TERM_CRITERIA_COUNT,
        10,
        0.03
    )
)
mask=np.zeros_like(old_frame)
while True:
    ret, frame=cap.read()
    if not ret:
        break
    frame_gray=cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
    new_points, status, error = cv2.calcOpticalFlowPyrLK(
        old_gray,
        frame_gray,
        old_points,
        None,
        **lk_params
    )
    if new_points is not None:
        good_new=new_points[status == 1]
        good_old=old_points[status == 1]
        for new, old in zip(good_new, good_old):
            x_new, y_new = new.ravel()
            x_old, y_old=old.ravel()
            mask=cv2.line(
                mask,
                (int(x_new),int(y_new)),
                (int(x_old),int(y_old)),
                (0,255,0),
                2
            )
            frame=cv2.circle(
                frame,
                (int(x_new),int(y_new)),
                5,
                (0,0,255),
                -1
            )
        output=cv2.add(frame,mask)
        cv2.imshow("Lucas-Kanade Optical flow",output)
        old_gray=frame_gray.copy()
        old_points=good_new.reshape(-1,1,2)
    if cv2.waitKey(30) & 0xFF== 27:
        break
cap.release()
cv2.destroyAllWindows()

Practical 8: Apply affine and perspective transforms to correct a document scan
import cv2
import numpy as np
image = cv2.imread(
    r"C:\Users\admin\OneDrive\Desktop\BSCAI-Practicals\Document.jpg"
)
if image is None:
    print("Image not found!")
    exit()
pts1 = np.float32([
    [100, 100],
    [500, 100],
    [100, 600]
])
pts2 = np.float32([
    [80, 120],
    [500, 100],
    [120, 600]
])
M = cv2.getAffineTransform(pts1, pts2)
affine = cv2.warpAffine(
    image,
    M,
    (image.shape[1], image.shape[0])
)
src = np.float32([
    [100, 80],
    [550, 100],
    [600, 650],
    [80, 620]
])
dst = np.float32([
    [0, 0],
    [500, 0],
    [500, 700],
    [0, 700]
])
P = cv2.getPerspectiveTransform(src, dst)
perspective = cv2.warpPerspective(
    image,
    P,
    (500, 700)
)
cv2.imshow("Original Document", image)
cv2.imshow("Affine Transformation", affine)
cv2.imshow("Perspective Correction", perspective)
cv2.imwrite("affine_document.jpg", affine)
cv2.imwrite("corrected_document.jpg", perspective)
cv2.waitKey(0)
cv2.destroyAllWindows()

Practical 9: Build a simple object detector using template matching
import cv2
image = cv2.imread(
    r"C:\Users\admin\OneDrive\Desktop\BSCAI-Practicals\Main Image.png"
)
template = cv2.imread(
    r"C:\Users\admin\OneDrive\Desktop\BSCAI-Practicals\template.jpg"
)
if image is None:
    print("Main image not found!")
    exit()
if template is None:
    print("Template image not found!")
    exit()
h, w = template.shape[:2]
result = cv2.matchTemplate(
    image,
    template,
    cv2.TM_CCOEFF_NORMED
)
min_val, max_val, min_loc, max_loc = cv2.minMaxLoc(result)
threshold = 0.60
if max_val >= threshold:
    top_left = max_loc
    bottom_right = (
        top_left[0] + w,
        top_left[1] + h
    )
    cv2.rectangle(
        image,
        top_left,
        bottom_right,
        (0, 255, 0),
        2
    )
    print("Object detected!")
    print("Matching score:", max_val)
else:
    print("Object not detected!")
    print("Best matching score:", max_val)
cv2.imshow("Object Detection", image)
cv2.waitKey(0)
cv2.destroyAllWindows()

Practical Extra 3: Perform image segmentation using thresholding and morphological operations
import cv2, numpy as np
import matplotlib.pyplot as plt
gray = cv2.imread("image.jpg", 0)
_, th = cv2.threshold(gray, 0, 255, cv2.THRESH_BINARY + cv2.THRESH_OTSU)
k = np.ones((5,5), np.uint8)
erosion = cv2.erode(th, k)
dilation = cv2.dilate(th, k)
opening = cv2.morphologyEx(th, cv2.MORPH_OPEN, k)
closing = cv2.morphologyEx(th, cv2.MORPH_CLOSE, k)
names = ["Original", "Threshold", "Erosion", "Dilation", "Opening", "Closing"]
for i, im in enumerate([gray, th, erosion, dilation, opening, closing]):
    plt.subplot(2, 3, i+1); plt.imshow(im, cmap="gray"); plt.title(names[i])
plt.show()

Practical Extra 4: Detect and Match image features using ORB and BFMatcher
import cv2
import matplotlib.pyplot as plt
img1 = cv2.imread("image1.jpg")
img2 = cv2.imread("image2.jpg")
orb = cv2.ORB_create()
k1, d1 = orb.detectAndCompute(cv2.cvtColor(img1, cv2.COLOR_BGR2GRAY), None)

bf = cv2.BFMatcher(cv2.NORM_HAMMING, crossCheck=True)
m = sorted(bf.match(d1, d2), key=lambda x: x.distance)
res = cv2.drawMatches(img1, k1, img2, k2, m[:50], None)
plt.imshow(cv2.cvtColor(res, cv2.COLOR_BGR2RGB)); plt.title("ORB + BFMatcher"); plt.axis("off")
plt.show()

Practical Extra 5: Implement image stitching using feature matching and homography
import cv2, numpy as np
import matplotlib.pyplot as plt
img1 = cv2.imread("left.jpg")
img2 = cv2.imread("right.jpg")
sift = cv2.SIFT_create()
k1, d1 = sift.detectAndCompute(cv2.cvtColor(img1, cv2.COLOR_BGR2GRAY), None)
k2, d2 = sift.detectAndCompute(cv2.cvtColor(img2, cv2.COLOR_BGR2GRAY), None)
matches = cv2.BFMatcher().knnMatch(d1, d2, k=2)
good = [m for m, n in matches if m.distance < 0.75 * n.distance]
src = np.float32([k2[m.trainIdx].pt for m in good]).reshape(-1,1,2)
dst = np.float32([k1[m.queryIdx].pt for m in good]).reshape(-1,1,2)
H, _ = cv2.findHomography(src, dst, cv2.RANSAC, 5.0)
result = cv2.warpPerspective(img2, H, (img1.shape[1] + img2.shape[1], img1.shape[0]))
result[0:img1.shape[0], 0:img1.shape[1]] = img1
plt.imshow(cv2.cvtColor(result, cv2.COLOR_BGR2RGB)); plt.title("Stitched Image"); plt.axis("off")
plt.show()
